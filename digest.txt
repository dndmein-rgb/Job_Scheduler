FROM node:20-alpine AS deps

COPY apps/worker/package.json ./apps/worker/

FROM node:20-alpine AS build
WORKDIR /app

COPY --from=deps /app/node_modules ./node_modules
COPY --from=deps /app/package.json ./package.json
COPY --from=deps /app/package-lock.json* ./

# Full source (node_modules excluded by .dockerignore)
COPY . .

# Critical: re-link workspaces now that full package sources exist
RUN npm install

# Optional but recommended for Prisma apps
ENV DATABASE_URL="postgresql://jobplatform:jobplatform@localhost:5432/jobplatform?schema=public"
RUN npx prisma generate --schema=packages/database/prisma/schema.prisma

RUN npm run build -w @job-platform/shared-types \
 && npm run build -w @job-platform/shared-events \
 && npm run build -w @job-platform/shared-kafka \
 && npm run build -w @job-platform/shared-utils \
 && npm run build -w @job-platform/database \
 && npm run build -w @job-platform/job-service \
 && npm run build -w @job-platform/worker \
 && npm run build -w @job-platform/scheduler \
 && npm run build -w @job-platform/reaper

FROM node:20-alpine AS runner
WORKDIR /app
ENV NODE_ENV=production

COPY --from=build /app/package.json ./

COPY --from=build /app/apps/scheduler ./apps/scheduler

CMD ["node", "apps/job-service/dist/server.js"]

type LogMeta = Record<string, unknown> | Error | unknown;

type LogFn = {
  (msg: string, meta?: LogMeta): void;
  (meta: LogMeta, msg?: string): void;
};

const toMergingObject = (meta?: LogMeta): Record<string, unknown> => {
  if (meta === undefined || meta === null) {
    return {};
  }

  if (meta instanceof Error) {
    return {
      err: {
        message: meta.message,
        name: meta.name,
        stack: meta.stack,
      },
    };
  }

  if (typeof meta === "object") {
    return meta as Record<string, unknown>;
  }

  // primitive (string/number/boolean) â€“ wrap it
  return { value: meta };
};

const createLogFn = (level: "info" | "error" | "warn" | "debug"): LogFn => {
  return (msgOrMeta: string | LogMeta, maybeMetaOrMsg?: string | LogMeta) => {
    // Preferred: message first
    // logger.error("Error during graceful shutdown", error)
    // logger.info("Member added", { roomId })
    if (typeof msgOrMeta === "string") {
      const msg = msgOrMeta;
      const obj = toMergingObject(maybeMetaOrMsg);
      baseLogger[level](obj, msg);
      return;
    }

    // Also support: object/error first
    // logger.error(error, "Error during graceful shutdown")
    // logger.info({ roomId }, "Member added")
    const obj = toMergingObject(msgOrMeta);
    const msg = typeof maybeMetaOrMsg === "string" ? maybeMetaOrMsg : undefined;

    if (msg) {
      baseLogger[level](obj, msg);
    } else {
      baseLogger[level](obj);
    }
  };
};

export const logger = {
  info: createLogFn("info"),
  error: createLogFn("error"),
  warn: createLogFn("warn"),
  debug: createLogFn("debug"),
  child: baseLogger.child.bind(baseLogger),
};

import { prisma } from "./prisma.js";
import { kafkaProducer } from "./kafka.js";
import { logger } from "../config/logger.js";

/**
 * Drains outbox â†’ Kafka.
 * At-least-once: crash after produce but before mark published may re-send.
 * Consumers should key off eventId.
 */
export class OutboxPublisher {
  private running = false;
  private timer: ReturnType<typeof setInterval> | null = null;
  private tickInFlight = false;

  start(pollIntervalMs = 1000): void {
    if (this.running) return;
    this.running = true;
    logger.info("OutboxPublisher started", { pollIntervalMs });
    this.timer = setInterval(() => {
      void this.tick();
    }, pollIntervalMs);
  }

  stop(): void {
    this.running = false;
    if (this.timer) {
      clearInterval(this.timer);
      this.timer = null;
    }
    logger.info("OutboxPublisher stopped");
  }

  private async tick(): Promise<void> {
    if (!this.running || this.tickInFlight) return;
    this.tickInFlight = true;
    try {
      const batch = await prisma.outbox.findMany({
        where: { publishedAt: null },
        orderBy: { createdAt: "asc" },
        take: 50,
      });

      for (const row of batch) {
        try {
          await kafkaProducer.send({
            topic: row.topic,
            messages: [
              {
                key: row.eventId,
                value: JSON.stringify(row.payload),
                headers: {
                  eventType: row.eventType,
                  eventId: row.eventId,
                },
              },
            ],
          });

          await prisma.outbox.update({
            where: { id: row.id },
            data: { publishedAt: new Date() },
          });
        } catch (err) {
          logger.error("Failed to publish outbox row", {
            outboxId: row.id,
            eventType: row.eventType,
            err,
          });
          // stop this batch; retry next tick
          break;
        }
      }
    } catch (err) {
      logger.error("OutboxPublisher tick failed", { err });
    } finally {
      this.tickInFlight = false;
    }
  }
}

export const outboxPublisher = new OutboxPublisher();

import { Redis } from "ioredis";
import { config } from "../config/index.js";
import { logger } from "../config/logger.js";

class RedisClient {
  private static instance: Redis | null = null;
  private static connected = false;

  private constructor() {}

  static getInstance(): Redis {
    if (RedisClient.instance) {
      return RedisClient.instance;
    }

    const client = new Redis(config.REDIS_URL, {
      retryStrategy: (times: number) => {
        const delay = Math.min(times * 500, 5000);

        logger.warn(
          { attempt: times, delayMs: delay },
          "Redis reconnect attempt",
        );

        return delay;
      },
      maxRetriesPerRequest: null,
      enableReadyCheck: true,
    });

    RedisClient.instance = client;
    RedisClient.setupEventListeners(client);

    return client;
  }

  private static setupEventListeners(client: Redis): void {
    client.on("connect", () => {
      logger.info("Redis TCP connection established");
    });

    client.on("ready", () => {
      RedisClient.connected = true;
      logger.info("Redis client is ready");
    });

    client.on("error", (error: Error) => {
      RedisClient.connected = false;
      logger.error({ err: error }, "Redis connection error");
    });

    client.on("close", () => {
      RedisClient.connected = false;
      logger.warn("Redis connection closed");
    });

    client.on("reconnecting", () => {
      RedisClient.connected = false;
      logger.warn("Reconnecting to Redis");
    });

    client.on("end", () => {
      RedisClient.connected = false;
      logger.warn("Redis connection ended");
    });
  }

  static isReady(): boolean {
    return RedisClient.connected;
  }

  static async testConnection(): Promise<boolean> {
    try {
      const response = await RedisClient.getInstance().ping();

      if (response === "PONG") {
        logger.info("Redis ping successful");
        return true;
      }

      return false;
    } catch (error) {
      logger.error({ err: error }, "Redis connection test failed");
      return false;
    }
  }

  static async closeConnection(): Promise<void> {
    const client = RedisClient.instance;

    if (!client) {
      return;
    }

    try {
      await client.quit();
      logger.info("Redis connection closed gracefully");
    } catch (error) {
      logger.error({ err: error }, "Error closing Redis connection");
      client.disconnect();
    } finally {
      RedisClient.instance = null;
      RedisClient.connected = false;
    }
  }
}

export const redis = RedisClient.getInstance();

export default RedisClient;

import type { Request, Response, NextFunction, RequestHandler } from "express";
import { AppError } from "@job-platform/shared-utils";
import { config } from "../config/index.js";

/**
 * Protect /admin routes with a shared API key.
 * Accepts: X-API-Key: <key>  OR  Authorization: Bearer <key>
 */
export const adminAuth: RequestHandler = (
  req: Request,
  _res: Response,
  next: NextFunction,
) => {
  const expected = (config as { ADMIN_API_KEY?: string }).ADMIN_API_KEY;

  // Dev escape: if no key configured, allow (log once in real apps)
  if (!expected) {
    next();
    return;
  }

  const headerKey = req.header("x-api-key");
  const auth = req.header("authorization");
  const bearer =
    auth && auth.toLowerCase().startsWith("bearer ")
      ? auth.slice(7).trim()
      : undefined;

  const provided = headerKey ?? bearer;

  if (!provided || provided !== expected) {
    next(
      new AppError("UNAUTHORIZED", "Invalid or missing admin API key", 401),
    );
    return;
  }

  next();
};

import type { Request, Response, NextFunction } from "express";
import { AppError, RateLimitError } from "@job-platform/shared-utils";
import { logger } from "../config/logger.js";

export function errorMiddleware(
  err: Error,
  _req: Request,
  res: Response,
  _next: NextFunction,
): void {

  if (err instanceof RateLimitError) {
      logger.warn(
        { err, code: err.code, retryAfterSec: err.retryAfterSec },
        err.message,
      );
      if (!res.getHeader("Retry-After")) {
        res.setHeader("Retry-After", String(err.retryAfterSec));
      }
      res.status(429).json({
        success: false,
        error: {
          code: err.code,
          message: err.message,
          details: err.details,
          retryAfterSec: err.retryAfterSec,
        },
      });
      return;
    }
  if (err instanceof AppError) {
    logger.warn({ err, code: err.code }, err.message);
    res.status(err.statusCode).json({
      success: false,
      error: {
        code: err.code,
        message: err.message,
        details: err.details,
      },
    });
    return;
  }

  logger.error({ err }, "Unhandled error");
  res.status(500).json({
    success: false,
    error: {
      code: "INTERNAL_ERROR",
      message:
        process.env.NODE_ENV === "production"
          ? "An unexpected error occurred"
          : err.message,
    },
  });
}

import rateLimit, {
  type Options,
  type ValueDeterminingMiddleware,
} from "express-rate-limit";
import { RedisStore, type RedisReply } from "rate-limit-redis";
import { Redis } from "ioredis";
import type { Request, Response, RequestHandler } from "express";
import { config } from "../config/index.js";
import { logger } from "../config/logger.js";
import { inc } from "../infrastructure/metrics.js";

/** Dedicated Redis client â€“ do NOT share BullMQâ€™s connection */
const rateLimitRedis = new Redis(config.REDIS_URL, {
  maxRetriesPerRequest: 3,
  enableReadyCheck: true,
});

rateLimitRedis.on("error", (err) => {
  logger.error("Rate-limit Redis error", { err });
});

function clientIp(req: Request): string {
  const xf = req.headers["x-forwarded-for"];
  if (typeof xf === "string" && xf.length > 0) {
    return xf.split(",")[0]!.trim();
  }
  // req.ip is set when "trust proxy" is enabled; fallback to socket
  return req.ip ?? req.socket.remoteAddress ?? "unknown";
}

interface RateLimiterOptions {
  windowMs: number;
  limit: number;
  prefix: string;
  error: { message: string; code: string };
  /** Must match express-rate-limit: (req, res) => string */
  keyGenerator?: ValueDeterminingMiddleware<string>;
}

export const createRateLimiter = ({
  windowMs,
  limit,
  prefix,
  error,
  keyGenerator,
}: RateLimiterOptions): RequestHandler => {
  const options: Partial<Options> = {
    windowMs,
    limit,
    standardHeaders: "draft-8",
    legacyHeaders: false,
    // Redis down â†’ allow traffic (readiness already reports redis fail)
    passOnStoreError: true,
    // Custom keyGenerator: disable IP-fallback validation warning
    validate: {
      keyGeneratorIpFallback: false,
    },
    store: new RedisStore({
      sendCommand: async (...args: string[]) => {
        const [command, ...commandArgs] = args;
        if (!command) {
          throw new Error("Redis command is required");
        }
        return rateLimitRedis.call(
          command,
          ...commandArgs,
        ) as Promise<RedisReply>;
      },
      prefix: `rate-limit:${prefix}:`,
    }),
    keyGenerator:
      keyGenerator ??
      (((req: Request, _res: Response) => {
        return clientIp(req);
      }) satisfies ValueDeterminingMiddleware<string>),
    handler: (_req, res) => {
      inc("jobs_rate_limited_total");
      res.status(429).json({
        success: false,
        error: {
          code: error.code,
          message: error.message,
        },
      });
    },
  };

  return rateLimit(options);
};

import type { Request, Response, NextFunction, RequestHandler } from "express";
import type { ZodSchema } from "zod";
import { ValidationError } from "@job-platform/shared-utils";

export function validateBody<T>(schema: ZodSchema<T>): RequestHandler {
  return (req: Request, _res: Response, next: NextFunction) => {
    const result = schema.safeParse(req.body);
    if (!result.success) {
      next(new ValidationError("Invalid request body", result.error.flatten()));
      return;
    }
    req.body = result.data;
    next();
  };
}

export function validateQuery<T>(schema: ZodSchema<T>): RequestHandler {
  return (req: Request, _res: Response, next: NextFunction) => {
    const result = schema.safeParse(req.query);
    if (!result.success) {
      next(new ValidationError("Invalid query parameters", result.error.flatten()));
      return;
    }
    (req as Request & { validatedQuery: T }).validatedQuery = result.data;
    next();
  };
}

/**
 * Admin / DLQ operations.
 * DEAD jobs are still rows in `jobs` with status = DEAD.
 * We do not introduce a separate DLQ table.
 */

import { JobStatus } from "@job-platform/shared-types";
import { JobInterface } from "../job/job.interface.js";
import { JobStats, ListDeadJobsFilter, ListDeadJobsResult } from "./admin.types.js";
import { Job, Prisma } from "../../infrastructure/prisma.js";
import { ConflictError, generateId, NotFoundError } from "@job-platform/shared-utils";
import { JOB_EVENT_TYPES } from "@job-platform/shared-events";
import { TOPICS } from "@job-platform/shared-kafka";
import { logger } from "../../config/logger.js";
import { enqueueJob, jobQueue } from "../../infrastructure/bullmq.js";

export class AdminService {
  constructor(private readonly jobRepo: JobInterface) {
    
  }

  /**
     * List only DEAD jobs (paginated + optional type/tenant filters).
     */

  async listDeadJobs(filter: ListDeadJobsFilter): Promise<ListDeadJobsResult> {
    return this.jobRepo.list({
      tenantId: filter.tenantId,
      status: JobStatus.DEAD,
      type: filter.type,
      page: filter.page,
      limit: filter.limit
    })
  }

  /**
     * Admin retry of a DEAD job â†’ QUEUED + enqueue.
     * Same semantics as JobService.retry, but restricted to DEAD.
     */
  async retryDead(id: string): Promise<Job> {
    const existing = await this.jobRepo.findById(id);
    if (!existing) throw new NotFoundError("Job", id);
    if (existing.status !== JobStatus.DEAD) {
      throw new ConflictError(
        `Admin dead-retry only allowed for DEAD jobs (current: ${existing.status})`,
        { status: existing.status },
      );
    }
    const eventId = generateId();
    const events = [
      {
        eventId,
        eventType: JOB_EVENT_TYPES.QUEUED,
        topic: TOPICS.JOB_EVENTS,
        payload: {
          eventId,
          eventType: JOB_EVENT_TYPES.QUEUED,
          occurredAt: new Date().toISOString(),
          jobId: id,
          jobType: existing.type,
          tenantId: existing.tenantId,
          payload: {
            status: JobStatus.QUEUED,
            priority: existing.priority,
            adminRetry: true,
            fromDead: true,
          },
        } as Prisma.InputJsonValue,
      },
    ];
    const retried = await this.jobRepo.updateStatusConditionalWithOutbox(id, [JobStatus.DEAD], {
      status: JobStatus.QUEUED,
      error: null,
      attempts: 0,
      leaseOwner: null,
      leaseExpiresAt: null,
      completedAt: null,
      cancelRequested: false,
    }, events)
    if (!retried) {
      throw new ConflictError(`Cannot retry job in status ${existing.status}`, {
        status: existing.status,
      });
    }

    try {
      await enqueueJob(id,existing.priority);
    } catch (err) {
      logger.error("Admin dead-retry: DB is QUEUED but enqueue failed", {
        jobId: id,
        err,
      });
    }
    
    logger.info("Dead job admin-retried", { jobId: id });
    return retried;
  }
  /**
     * Permanently close a DEAD job â†’ CANCELLED.
     * Requires DEAD â†’ CANCELLED in the state machine.
     */

  async cancelDead(id: string, reason?: string | null): Promise<Job> {
    const existing = await this.jobRepo.findById(id);
    if (!existing) throw new NotFoundError("Job", id);
    
    if (existing.status !== JobStatus.DEAD) {
      throw new ConflictError(
        `Admin dead-cancel only allowed for DEAD jobs (current: ${existing.status})`,
        { status: existing.status },
      );
    }
    
    const eventId = generateId();
    const events = [
      {
        eventId,
        eventType: JOB_EVENT_TYPES.CANCELLED,
        topic: TOPICS.JOB_EVENTS,
        payload: {
          eventId,
          eventType: JOB_EVENT_TYPES.CANCELLED,
          occurredAt: new Date().toISOString(),
          jobId: id,
          jobType: existing.type,
          tenantId: existing.tenantId,
          payload: {
            reason: reason ?? "admin_dead_cancel",
            fromDead: true,
          },
        } as Prisma.InputJsonValue,
      },
    ];
    
    const cancelled = await this.jobRepo.updateStatusConditionalWithOutbox(
      id,
      [JobStatus.DEAD],
      {
        status: JobStatus.CANCELLED,
        cancelRequested: true,
        leaseOwner: null,
        leaseExpiresAt: null,
      },
      events,
    );
    
    if (!cancelled) {
      throw new ConflictError(
        `Cannot cancel dead job in status ${existing.status}`,
        { status: existing.status },
      );
    }
    
    logger.info("Dead job cancelled by admin", {
      jobId: id,
      reason: reason ?? null,
    });
    return cancelled;
  }
  


  /**
     * Simple aggregate counts by status (source of truth = Postgres).
     */

  async getStats(): Promise<JobStats>{
    const byStatus = await this.jobRepo.countByStatus();
    
        const get = (s: string) => byStatus[s] ?? 0;
        const total = Object.values(byStatus).reduce((sum, n) => sum + n, 0);
    
        return {
          byStatus,
          total,
          dead: get(JobStatus.DEAD),
          running: get(JobStatus.RUNNING),
          queued: get(JobStatus.QUEUED),
          failed: get(JobStatus.FAILED),
          scheduled: get(JobStatus.SCHEDULED),
          completed: get(JobStatus.COMPLETED),
          cancelled: get(JobStatus.CANCELLED),
          retrying: get(JobStatus.RETRYING),
        };
  }


  async getQueueStats(): Promise<{
    queue:string,counts:Record<string,number>
  }>{
    const counts=await   jobQueue.getJobCounts("waiting",
          "active",
          "completed",
          "failed",
          "delayed",
          "paused",
        );
        return { queue: "jobs", counts };
  }
  async getOutboxStats() {
    const pending = await this.jobRepo.countOutboxPending();
    return { pending };
  }
  
}

import { generateId } from "@job-platform/shared-utils";
import { Job, Prisma, prisma } from "../../infrastructure/prisma.js";
import { JobInterface, JobStatusCounts } from "./job.interface.js";
import {
  CreateJobWriteData,
  ListJobsFilter,
  ListJobsResult,
  OutboxEventInput,
} from "./job.types.js";

export class JobRepository implements JobInterface {
  async findById(id: string): Promise<Job | null> {
    return prisma.job.findUnique({
      where: { id },
    });
  }
  async findByIdempotencyKey(key: string): Promise<Job | null> {
    return prisma.job.findUnique({
      where: { idempotencyKey: key },
    });
  }
  /**
   * Create job + outbox row(s) atomically.
   * Why one transaction? Outbox pattern â€” no dual-write with Kafka.
   */
  async createWithOutbox(
    data: CreateJobWriteData,
    events: OutboxEventInput[],
  ): Promise<Job> {
    
    return prisma.$transaction(async (tx) => {
      const job = await tx.job.create({
        data: {
          id:data.id,
          type: data.type,
          status:data.status,
          priority: data.priority ?? "MEDIUM",
          payload: data.payload,
          maxAttempts: data.maxAttempts??3,
          scheduledAt: data.scheduledAt,
          idempotencyKey: data.idempotencyKey ?? null,
          tenantId: data.tenantId ?? null,
          attempts: 0,
          cancelRequested: false,
        },
      });
      if (events.length > 0) {
        await tx.outbox.createMany({
          data: events.map((e) => ({
            id: generateId(),
            eventId: e.eventId,
            eventType: e.eventType,
            topic: e.topic,
            payload: e.payload,
            publishedAt: null,
          })),
        });
      }
      return job;
    });
  }

  async list(filter: ListJobsFilter): Promise<ListJobsResult> {
    const page = Math.max(1, filter.page ?? 1);
    const limit = Math.min(100, Math.max(1, filter.limit ?? 20));
    const skip = (page - 1) * limit;
    const where: Prisma.JobWhereInput = {};
    if (filter.status) where.status = filter.status;
    if (filter.type) where.type = filter.type;
    if (filter.tenantId) where.tenantId = filter.tenantId;

    const [items, total] = await Promise.all([
      prisma.job.findMany({
        where,
        orderBy: { createdAt: "desc" },
        skip,
        take: limit,
      }),
      prisma.job.count({ where }),
    ]);
    return {
      items,
      total,
      page,
      limit,
      totalPages: Math.ceil(total / limit) || 1,
    };
  }

  /**
   * Status update with optional outbox events (used from Phase 2+).
   * Service must call assertTransition before this.
   */

  async updateStatusWithOutbox(
    id: string,
    data: Prisma.JobUpdateInput,
    events: OutboxEventInput[],
  ): Promise<Job> {
    return prisma.$transaction(async (tx) => {
      const job = await tx.job.update({
        where: { id },
        data,
      });
      if (events.length > 0) {
        await tx.outbox.createMany({
          data: events.map((e) => ({
            id: generateId(),
            eventId: e.eventId,
            eventType: e.eventType,
            topic: e.topic,
            payload: e.payload,
            publishedAt: null,
          })),
        });
      }
      return job;
    });
  }
  /**
   * Optimistic status transition.
   * Returns the updated job, or null if no row matched (lost race / wrong status).
   */
  async updateStatusConditionalWithOutbox(
    id: string,
    fromStatuses: string[],
    data: Prisma.JobUpdateInput,
    events: OutboxEventInput[],
  ): Promise<Job | null> {
    return prisma.$transaction(async (tx) => {
      const result = await tx.job.updateMany({
        where: {
          id,
          status: { in: fromStatuses as any },
        },
        data,
      });
  
      if (result.count === 0) {
        return null;
      }
  
      if (events.length > 0) {
        await tx.outbox.createMany({
          data: events.map((e) => ({
            id: generateId(),
            eventId: e.eventId,
            eventType: e.eventType,
            topic: e.topic,
            payload: e.payload,
            publishedAt: null,
          })),
        });
      }
  
      return tx.job.findUniqueOrThrow({ where: { id } });
    });
  }
  async countByStatus(): Promise<JobStatusCounts> {
    const groups = await prisma.job.groupBy({
      by: ["status"],
      _count: { _all: true },
    })
    const counts: Record<string, number> = {}
    for (const row of groups) {
      counts[row.status]=row._count._all
    }
    return counts;
  }

  async countOutboxPending(): Promise<number> {
    return prisma.outbox.count({ where: { publishedAt: null } });
  }
}

import { JOB_EVENT_TYPES } from "@job-platform/shared-events";
import { TOPICS } from "@job-platform/shared-kafka";
import type { Job, Prisma } from "../../infrastructure/prisma.js";
import type { JobInterface } from "./job.interface.js";
import type { CreateJobInput } from "./job.schema.js";
import type {
  CreateJobWriteData,
  ListJobsFilter,
  ListJobsResult,
  OutboxEventInput,
} from "./job.types.js";
import { initialStatusForCreate } from "./job.state-machine.js";
import { logger } from "../../config/logger.js";
import { enqueueJob } from "../../infrastructure/bullmq.js";
import { JobStatus } from "@job-platform/shared-types";

export class JobService {
  constructor(private readonly repo: JobInterface) {}

  async create(input: CreateJobInput): Promise<Job> {
    if (input.idempotencyKey) {
      const existing = await this.repo.findByIdempotencyKey(input.idempotencyKey);
      if (existing) {
        logger.info("Idempotent create â€“ returning existing job", {
          jobId: existing.id,
          idempotencyKey: input.idempotencyKey,
        });
        return existing;
      }
    }

    const id = generateId();
    const status = initialStatusForCreate(input.scheduledAt ?? null);
    const priority = input.priority ?? "MEDIUM";
    const maxAttempts = input.maxAttempts ?? 3;
    const eventId = generateId();

    const write: CreateJobWriteData = {
      id,
      type: input.type,
      status,
      priority,
      payload: input.payload as Prisma.InputJsonValue,
      maxAttempts,
      scheduledAt: input.scheduledAt ?? null,
      idempotencyKey: input.idempotencyKey ?? null,
      tenantId: input.tenantId ?? null,
    };

    const events: OutboxEventInput[] = [
      {
        eventId,
        eventType: JOB_EVENT_TYPES.CREATED,
        topic: TOPICS.JOB_EVENTS,
        payload: {
          eventId,
          eventType: JOB_EVENT_TYPES.CREATED,
          occurredAt: new Date().toISOString(),
          jobId: id,
          jobType: input.type,
          tenantId: input.tenantId ?? null,
          payload: {
            status,
            priority,
            scheduledAt: input.scheduledAt?.toISOString() ?? null,
            idempotencyKey: input.idempotencyKey ?? null,
            maxAttempts,
          },
        },
      },
    ];

    const job = await this.repo.createWithOutbox(write, events);

    logger.info("Job created", {
      jobId: job.id,
      status: job.status,
      type: job.type,
    });
    if (job.status === "QUEUED") {
      try {
        await enqueueJob(job.id,job.priority)
      } catch (err) {
        // Job is durable in DB; missing queue message is recoverable later
            logger.error("Failed to enqueue job (DB row exists as QUEUED)", {
              jobId: job.id,
              err,
            });
      }
    }

    return job;
  }

  async getById(id: string): Promise<Job> {
    const job = await this.repo.findById(id);
    if (!job) throw new NotFoundError("Job", id);
    return job;
  }

  async list(filter: ListJobsFilter): Promise<ListJobsResult> {
    return this.repo.list(filter);
  }
  /**
   * Cancel a job.
   * Allowed: PENDING | SCHEDULED | QUEUED | RETRYING | RUNNING â†’ CANCELLED
   */
  async cancel(id: string, reason?: string | null): Promise<Job>{
    const fromStatuses: JobStatus[] = [
      JobStatus.PENDING,
      JobStatus.SCHEDULED,
      JobStatus.RETRYING,
      JobStatus.QUEUED,
      JobStatus.RUNNING
    ]
    const eventId = generateId();
    const events = [
        {
          eventId,
          eventType: JOB_EVENT_TYPES.CANCELLED,
          topic: TOPICS.JOB_EVENTS,
          payload: {
            eventId,
            eventType: JOB_EVENT_TYPES.CANCELLED,
            occurredAt: new Date().toISOString(),
            jobId: id,
            jobType: "", // filled after load if you prefer; or pass from pre-read
            tenantId: null,
            payload: { reason: reason ?? null },
          } as Prisma.InputJsonValue,
        },
      ];
    
      
    // Optional: pre-read for jobType/tenantId in the event
      const existing = await this.repo.findById(id);
    if (!existing) throw new NotFoundError("Job", id);

    events[0]!.payload = {
        eventId,
        eventType: JOB_EVENT_TYPES.CANCELLED,
        occurredAt: new Date().toISOString(),
        jobId: id,
        jobType: existing.type,
        tenantId: existing.tenantId,
        payload: { reason: reason ?? null },
      } as Prisma.InputJsonValue;
    const cancelled = await this.repo.updateStatusConditionalWithOutbox(
        id,
        fromStatuses,
        {
          status: JobStatus.CANCELLED,
          cancelRequested: true,
          leaseOwner: null,
          leaseExpiresAt: null,
        },
        events,
      );
    
      if (!cancelled) {
        throw new ConflictError(`Cannot cancel job in status ${existing.status}`, {
          status: existing.status,
        });
      }
    
      logger.info("Job cancelled", { jobId: id, reason: reason ?? null });
      return cancelled;
  }
  async retry(id: string): Promise<Job> {
    const existing = await this.repo.findById(id);
    if (!existing) throw new NotFoundError("Job", id);
  
    const fromStatuses = [JobStatus.FAILED, JobStatus.DEAD];
    const eventId = generateId();
  
    const events = [
      {
        eventId,
        eventType: JOB_EVENT_TYPES.QUEUED,
        topic: TOPICS.JOB_EVENTS,
        payload: {
          eventId,
          eventType: JOB_EVENT_TYPES.QUEUED,
          occurredAt: new Date().toISOString(),
          jobId: id,
          jobType: existing.type,
          tenantId: existing.tenantId,
          payload: {
            status: JobStatus.QUEUED,
            priority: existing.priority,
            adminRetry: true,
          },
        } as Prisma.InputJsonValue,
      },
    ];
  
    const retried = await this.repo.updateStatusConditionalWithOutbox(
      id,
      fromStatuses,
      {
        status: JobStatus.QUEUED,
        error: null,
        attempts: 0,
        leaseOwner: null,
        leaseExpiresAt: null,
        completedAt: null,
        cancelRequested: false,
      },
      events,
    );
  
    if (!retried) {
      throw new ConflictError(`Cannot retry job in status ${existing.status}`, {
        status: existing.status,
      });
    }
  
    try {
      await enqueueJob(id,existing.priority);
    } catch (err) {
      logger.error("Admin retry: DB is QUEUED but enqueue failed", { jobId: id, err });
    }
  
    logger.info("Job admin-retried", { jobId: id });
    return retried;
  }
}

/**
 * Explicit lifecycle graph.
 * Why? Prevents illegal updates and documents the system in one place.
 */

import { JobStatus } from "@job-platform/shared-types";
import { InvalidStateTransitionError } from "@job-platform/shared-utils";

const ALLOWED_TRANSITIONS: Record<JobStatus, readonly JobStatus[]> = {
  PENDING: [JobStatus.QUEUED, JobStatus.SCHEDULED, JobStatus.CANCELLED],
  SCHEDULED: [JobStatus.QUEUED, JobStatus.CANCELLED],
  QUEUED: [JobStatus.RUNNING, JobStatus.CANCELLED],
  RUNNING: [
    JobStatus.COMPLETED,
    JobStatus.FAILED,
    JobStatus.CANCELLED,
    JobStatus.QUEUED, // reaper lease recovery only
  ],
  FAILED: [JobStatus.RETRYING, JobStatus.DEAD,JobStatus.QUEUED],// admin retry
  RETRYING: [JobStatus.QUEUED,JobStatus.CANCELLED],
  COMPLETED: [],
  DEAD: [JobStatus.QUEUED,JobStatus.CANCELLED],// admin retry
  CANCELLED: [],
};

export function canTransition(from: JobStatus, to: JobStatus): boolean {
  return ALLOWED_TRANSITIONS[from].includes(to);
}

export function assertTransition(from: JobStatus, to: JobStatus): void {
  if (!canTransition(from, to)) {
    throw new InvalidStateTransitionError(from, to);
  }
}

export function initialStatusForCreate(
  scheduledAT: Date | null | undefined,
): JobStatus {
  if (scheduledAT && scheduledAT.getTime() > Date.now()) {
    return JobStatus.SCHEDULED;
  }
  return JobStatus.QUEUED;
}

});

let timer: ReturnType<typeof setInterval> | null = null;
let tickInFlight = false;

async function tick() {
  if (tickInFlight) return;
  tickInFlight = true;
  try {
    const result = await reaperService.tick();
    if (result.jobsRecovered > 0 || result.workersMarkedOffline > 0) {
      logger.info("Reaper tick", result);
    }
  } catch (err) {
    logger.error("Reaper tick failed", { err });
  } finally {
    tickInFlight = false;
  }
}

timer = setInterval(() => void tick(), config.REAPER_POLL_INTERVAL_MS);
// Run once immediately so we don't wait a full interval after boot.
void tick();

  logger.info("Shutting down reaperâ€¦", { signal });
  if (timer) {
    clearInterval(timer);
    timer = null;
  }
  process.exit(0);
}

process.on("SIGTERM", () => void shutdown("SIGTERM"));
process.on("SIGINT", () => void shutdown("SIGINT"));

import { config } from "./index.js";

const baseLogger = pino({
  level: config.NODE_ENV === "development" ? "debug" : "info",
  ...(config.NODE_ENV === "development" && {
    transport: {
      target: "pino-pretty",
      options: {
        colorize: true,
        translateTime: "SYS:standard",
        ignore: "pid,hostname",
      },
    },
  }),
  base: {
    service: config.SERVICE_NAME,
  },
});

type LogMeta = Record<string, unknown> | Error | unknown;

type LogFn = {
  (msg: string, meta?: LogMeta): void;
  (meta: LogMeta, msg?: string): void;
};

const toMergingObject = (meta?: LogMeta): Record<string, unknown> => {
  if (meta === undefined || meta === null) {
    return {};
  }

  if (meta instanceof Error) {
    return {
      err: {
        message: meta.message,
        name: meta.name,
        stack: meta.stack,
      },
    };
  }

  if (typeof meta === "object") {
    return meta as Record<string, unknown>;
  }

  // primitive (string/number/boolean) â€“ wrap it
  return { value: meta };
};

const createLogFn = (level: "info" | "error" | "warn" | "debug"): LogFn => {
  return (msgOrMeta: string | LogMeta, maybeMetaOrMsg?: string | LogMeta) => {
    // Preferred: message first
    // logger.error("Error during graceful shutdown", error)
    // logger.info("Member added", { roomId })
    if (typeof msgOrMeta === "string") {
      const msg = msgOrMeta;
      const obj = toMergingObject(maybeMetaOrMsg);
      baseLogger[level](obj, msg);
      return;
    }

    // Also support: object/error first
    // logger.error(error, "Error during graceful shutdown")
    // logger.info({ roomId }, "Member added")
    const obj = toMergingObject(msgOrMeta);
    const msg = typeof maybeMetaOrMsg === "string" ? maybeMetaOrMsg : undefined;

    if (msg) {
      baseLogger[level](obj, msg);
    } else {
      baseLogger[level](obj);
    }
  };
};

export const logger = {
  info: createLogFn("info"),
  error: createLogFn("error"),
  warn: createLogFn("warn"),
  debug: createLogFn("debug"),
  child: baseLogger.child.bind(baseLogger),
};

import { Queue } from "bullmq";
import { redis } from "./redis.js";
import { logger } from "../config/logger.js";
import { DomainPriority, mapPriorityToBullmq } from "@job-platform/shared-utils";

export const JOB_QUEUE_NAME = "jobs";
export const JOB_PROCESS_NAME = "process";

export type QueueJobData = { jobId: string };

export const jobQueue = new Queue<QueueJobData>(JOB_QUEUE_NAME, {
  connection: redis,
  defaultJobOptions: {
    removeOnComplete: { count: 1000 },
    removeOnFail: { count: 5000 },
    attempts: 1,
  },
});

export async function enqueueJob(
  jobId: string,
  priority?: DomainPriority,
): Promise<void> {
  const bullPriority = mapPriorityToBullmq(priority);
  await jobQueue.add(
    JOB_PROCESS_NAME,
    { jobId },
    {
      // stable id â†’ at-most-once enqueue for same jobId (BullMQ dedupe by jobId)
      jobId,
      priority: bullPriority,
    },
  );
  logger.info("Reaper re-enqued job", {
    jobId,
    priority: priority ?? "MEDIUM",
    bullPriority,
  });
}

 * Why a grace period?
 * A worker may be mid-renewal under DB/network pressure. We wait a little
 * past leaseExpiresAt so we don't steal a live job.
 *
 * Transition: RUNNING â†’ QUEUED (allowed only for reaper path).
 * We clear the lease and re-enqueue so another worker can claim it.
 */
export class ReaperService {

  async tick(): Promise<{ jobsRecovered: number; workersMarkedOffline: number }> {
      const jobsRecovered = await this.reapExpiredLeases();
      const workersMarkedOffline = await this.markStaleWorkersOffline();
      return { jobsRecovered, workersMarkedOffline };
    }
  private async reapExpiredLeases(): Promise<number> {
    const cutoff = new Date(Date.now() - config.LEASE_GRACE_MS);

    const expired = await prisma.job.findMany({
      where: {
        status: "RUNNING",
        leaseExpiresAt: { lt: cutoff },
      },
      take: 50,
      orderBy: { leaseExpiresAt: "asc" },
    });

    if (expired.length === 0) return 0;

    let recovered = 0;

    for (const job of expired) {
      try {
        const eventId = generateId();
        const previousWorkerId = job.leaseOwner ?? "unknown";
        const leaseExpiredAt =
          job.leaseExpiresAt?.toISOString() ?? new Date().toISOString();
        // Returns true only when we actually won the race and wrote the outbox row.
        const didRecover = await prisma.$transaction(async (tx) => {
          // Optimistic: only recover if still RUNNING with the same expired lease.
          const updated = await tx.job.updateMany({
            where: {
              id: job.id,
              status: "RUNNING",
              leaseExpiresAt: { lt: cutoff },
            },
            data: {
              status: "QUEUED",
              leaseOwner: null,
              leaseExpiresAt: null,
            },
          });

          if (updated.count === 0) {
            // Lost the race (worker finished or another reaper instance).
            return false;
          }

          await tx.outbox.create({
            data: {
              id: generateId(),
              eventId,
              eventType: JOB_EVENT_TYPES.LEASE_EXPIRED,
              topic: TOPICS.JOB_EVENTS,
              payload: {
                eventId,
                eventType: JOB_EVENT_TYPES.LEASE_EXPIRED,
                occurredAt: new Date().toISOString(),
                jobId: job.id,
                jobType: job.type,
                tenantId: job.tenantId,
                payload: {
                  previousWorkerId,
                  leaseExpiredAt,
                },
              },
            },
          });
          return true;
        });
        if (!didRecover) {
          // Someone else finished/reaped it â€” do not enqueue or count.
          continue;
        }
        // Re-enqueue after the DB commit (same pattern as job-service create).
        try {
          await enqueueJob(job.id,job.priority);
        } catch (err) {
          logger.error(
            "Reaper: DB recovered but enqueue failed (job is QUEUED)",
            {
              jobId: job.id,
              err,
            },
          );
        }

        recovered += 1;
        logger.warn("Lease expired â€“ job requeued", {
          jobId: job.id,
          previousWorkerId,
          leaseExpiredAt,
        });
      } catch (err) {
        logger.error("Reaper failed to recover job", { jobId: job.id, err });
      }
    }

    return recovered;
  }
  /**
     * ONLINE workers whose heartbeat is older than WORKER_STALE_MS â†’ OFFLINE.
     * Why? Crash / kill -9 never runs markOffline(); the table would lie forever.
     */
    private async markStaleWorkersOffline(): Promise<number> {
      const cutoff = new Date(Date.now() - config.WORKER_STALE_MS);
  
      const stale = await prisma.worker.findMany({
        where: {
          status: "ONLINE",
          lastHeartbeatAt: { lt: cutoff },
        },
        take: 50,
        orderBy: { lastHeartbeatAt: "asc" },
      });
  
      if (stale.length === 0) return 0;
  
      let marked = 0;
  
      for (const w of stale) {
        try {
          const eventId = generateId();
          const didMark = await prisma.$transaction(async (tx) => {
            const updated = await tx.worker.updateMany({
              where: {
                id: w.id,
                status: "ONLINE",
                lastHeartbeatAt: { lt: cutoff },
              },
              data: {
                status: "OFFLINE",
              },
            });
  
            if (updated.count === 0) return false;
  
            await tx.outbox.create({
              data: {
                id: generateId(),
                eventId,
                eventType: WORKER_EVENT_TYPES.OFFLINE,
                topic: TOPICS.WORKER_EVENTS,
                payload: {
                  eventId,
                  eventType: WORKER_EVENT_TYPES.OFFLINE,
                  occurredAt: new Date().toISOString(),
                  workerId: w.id,
                  payload: {
                    reason: "stale_heartbeat",
                    lastHeartbeatAt: w.lastHeartbeatAt.toISOString(),
                  },
                },
              },
            });
  
            return true;
          });
  
          if (didMark) {
            marked += 1;
            logger.warn("Worker marked offline (stale heartbeat)", {
              workerId: w.id,
              lastHeartbeatAt: w.lastHeartbeatAt.toISOString(),
            });
          }
        } catch (err) {
          logger.error("Failed to mark stale worker offline", {
            workerId: w.id,
            err,
          });
        }
      }
  
      return marked;
    }
}


export const reaperService = new ReaperService();

  env: config.NODE_ENV,
});

let timer: ReturnType<typeof setInterval> | null = null;
let tickInFlight = false;

async function tick() {
  if (tickInFlight) return;
  tickInFlight = true;
  try {
    const promoted = await schedulerService.tick();
    if (promoted > 0) {
      logger.info("Scheduler tick promoted jobs", { promoted });
    }
  } catch (err) {
    logger.error("Scheduler tick failed", { err });
  } finally {
    tickInFlight = false;
  }
}

timer = setInterval(() => void tick(), config.SCHEDULER_POLL_INTERVAL_MS);
// Run once immediately so we don't wait a full interval after boot.
void tick();

async function shutdown(signal: string) {
  logger.info("Shutting down schedulerâ€¦", { signal });
  if (timer) {
    clearInterval(timer);
    timer = null;
  }
  process.exit(0);
}

process.on("SIGTERM", () => void shutdown("SIGTERM"));
process.on("SIGINT", () => void shutdown("SIGINT"));

    service: config.SERVICE_NAME,
  },

type LogMeta = Record<string, unknown> | Error | unknown;

type LogFn = {
  (msg: string, meta?: LogMeta): void;
  (meta: LogMeta, msg?: string): void;
};

const toMergingObject = (meta?: LogMeta): Record<string, unknown> => {
  if (meta === undefined || meta === null) {
    return {};
  }

  if (meta instanceof Error) {
    return {
      err: {
        message: meta.message,
        name: meta.name,
        stack: meta.stack,
      },
    };
  }

  if (typeof meta === "object") {
    return meta as Record<string, unknown>;
  }

  // primitive (string/number/boolean) â€“ wrap it
  return { value: meta };
};

const createLogFn = (level: "info" | "error" | "warn" | "debug"): LogFn => {
  return (msgOrMeta: string | LogMeta, maybeMetaOrMsg?: string | LogMeta) => {
    // Preferred: message first
    // logger.error("Error during graceful shutdown", error)
    // logger.info("Member added", { roomId })
    if (typeof msgOrMeta === "string") {
      const msg = msgOrMeta;
      const obj = toMergingObject(maybeMetaOrMsg);
      baseLogger[level](obj, msg);
      return;
    }

    // Also support: object/error first
    // logger.error(error, "Error during graceful shutdown")
    // logger.info({ roomId }, "Member added")
    const obj = toMergingObject(msgOrMeta);
    const msg = typeof maybeMetaOrMsg === "string" ? maybeMetaOrMsg : undefined;

    if (msg) {
      baseLogger[level](obj, msg);
    } else {
      baseLogger[level](obj);
    }
  };
};

export const logger = {
  info: createLogFn("info"),
  error: createLogFn("error"),
  warn: createLogFn("warn"),
  debug: createLogFn("debug"),
  child: baseLogger.child.bind(baseLogger),
};

import { Queue } from "bullmq";
import { redis } from "./redis.js";
import { logger } from "../config/logger.js";
import { DomainPriority, mapPriorityToBullmq } from "@job-platform/shared-utils";

export const JOB_QUEUE_NAME = "jobs";
export const JOB_PROCESS_NAME = "process";

export type QueueJobData = { jobId: string };

export const jobQueue = new Queue<QueueJobData>(JOB_QUEUE_NAME, {
  connection: redis,
  defaultJobOptions: {
    removeOnComplete: { count: 1000 },
    removeOnFail: { count: 5000 },
    attempts: 1,
  },
});

export async function enqueueJob(
  jobId: string,
  priority?: DomainPriority,
): Promise<void> {
  const bullPriority = mapPriorityToBullmq(priority);
  await jobQueue.add(
    JOB_PROCESS_NAME,
    { jobId },
    {
      // stable id â†’ at-most-once enqueue for same jobId (BullMQ dedupe by jobId)
      jobId,
      priority: bullPriority,
    },
  );
  logger.info("Scheduler enqueued job to BullMQ", {
    jobId,
    priority: priority ?? "MEDIUM",
    bullPriority,
  });
}

import { generateId } from "@job-platform/shared-utils";
import { JOB_EVENT_TYPES } from "@job-platform/shared-events";
import { TOPICS } from "@job-platform/shared-kafka";
import { prisma } from "../infrastructure/prisma.js";
import { enqueueJob } from "../infrastructure/bullmq.js";
import { logger } from "../config/logger.js";

/**
 * Promote due SCHEDULED jobs â†’ QUEUED + enqueue.
 *
 * Why optimistic update?
 * Multiple scheduler instances (or a concurrent cancel) must not double-promote.
 * We only enqueue when the DB transition actually succeeded.
 */
export class SchedulerService {
  async tick(): Promise<number> {
    const now = new Date();

    const due = await prisma.job.findMany({
      where: {
        status: "SCHEDULED",
        scheduledAt: { lte: now },
      },
      take: 50,
      orderBy: { scheduledAt: "asc" },
    });

    if (due.length === 0) return 0;

    let promoted = 0;

    for (const job of due) {
      try {
        const eventId = generateId();

        const didPromote = await prisma.$transaction(async (tx) => {
          const updated = await tx.job.updateMany({
            where: {
              id: job.id,
              status: "SCHEDULED",
            },
            data: {
              status: "QUEUED",
            },
          });

          if (updated.count === 0) {
            // Lost the race (cancelled, or another scheduler instance).
            return false;
          }

          await tx.outbox.create({
            data: {
              id: generateId(),
              eventId,
              eventType: JOB_EVENT_TYPES.QUEUED,
              topic: TOPICS.JOB_EVENTS,
              payload: {
                eventId,
                eventType: JOB_EVENT_TYPES.QUEUED,
                occurredAt: new Date().toISOString(),
                jobId: job.id,
                jobType: job.type,
                tenantId: job.tenantId,
                payload: {
                  status: "QUEUED",
                  priority: job.priority,
                  scheduledAt: job.scheduledAt?.toISOString() ?? null,
                },
              },
            },
          });

          return true;
        });

        if (!didPromote) {
          continue;
        }

        try {
          await enqueueJob(job.id);
        } catch (err) {
          logger.error(
            "Scheduler: DB promoted but enqueue failed (job is QUEUED)",
            { jobId: job.id, err },
          );
        }

        promoted += 1;
        logger.info("Scheduled job promoted to QUEUED", {
          jobId: job.id,
          scheduledAt: job.scheduledAt?.toISOString(),
        });
      } catch (err) {
        logger.error("Scheduler failed to promote job", { jobId: job.id, err });
      }
    }

    return promoted;
  }
}

export const schedulerService = new SchedulerService();

const BASE = import.meta.env.VITE_API_BASE_URL ?? "http://localhost:3010";
const ADMIN_KEY = import.meta.env.VITE_ADMIN_API_KEY ?? "";

type ApiOk<T> = { success: true; data: T; meta?: Record<string, unknown> };
type ApiErr = {
  success: false;
  error: { code: string; message: string; details?: unknown };
};

export async function apiFetch<T = unknown>(
  path: string,
  options: RequestInit & { admin?: boolean } = {},
): Promise<T> {
  const { admin, ...init } = options;
  const headers = new Headers(init.headers);
  if (!headers.has("Content-Type") && init.body) {
    headers.set("Content-Type", "application/json");
  }
  if (admin && ADMIN_KEY) headers.set("X-API-Key", ADMIN_KEY);

  const res = await fetch(`${BASE}${path}`, { ...init, headers });
  const json = await res.json();

  if (!res.ok) {
    throw new Error(json?.error?.message ?? `HTTP ${res.status}`);
  }
  return json as T;
}
async function request<T>(
  path: string,
  init: RequestInit = {},
  admin = false,
): Promise<T> {
  const headers = new Headers(init.headers);
  headers.set("Content-Type", "application/json");
  if (admin && ADMIN_KEY) {
    headers.set("X-API-Key", ADMIN_KEY);
  }

  const res = await fetch(`${BASE}${path}`, { ...init, headers });
  const body = (await res.json()) as ApiOk<T> | ApiErr;

  if (!res.ok || body.success === false) {
    const msg =
      body && "error" in body
        ? body.error.message
        : `HTTP ${res.status}`;
    throw new Error(msg);
  }

  return body.data;
}

export const api = {

      return res.json();
    }),

  getJob: (id: string) => request<Record<string, unknown>>(`/jobs/${id}`),

  createJob: (body: unknown) =>

    }),

  // admin

import { Worker } from "bullmq";
import { config } from "./config/index.js";
import { logger } from "./config/logger.js";
import { redis } from "./infrastructure/redis.js";
import { prisma } from "./infrastructure/prisma.js";
import { jobLifecycle } from "./services/job-lifecycle.service.js";
import { getProcessor } from "./processors/index.js";
import { startLeaseHeartbeat } from "./services/lease-heartbeat.js";
import { workerRegistry } from "./services/worker-registry.service.js";

const JOB_QUEUE_NAME = "jobs";

async function main() {
  await workerRegistry.register();
  workerRegistry.startHeartbeat();

  logger.info("Worker starting", {
    workerId: config.WORKER_ID,
    capabilities: config.WORKER_CAPABILITIES,
    leaseMs: config.LEASE_MS,
    env: config.NODE_ENV,
  });

  const worker = new Worker<{ jobId: string }>(
    JOB_QUEUE_NAME,
    async (bullJob) => {
      const jobId = bullJob.data.jobId;
      const t0 = Date.now();

      const claimed = await jobLifecycle.claim(jobId);
      const stopHeartbeat = startLeaseHeartbeat(jobId);

      try {
        const latest = await prisma.job.findUnique({ where: { id: jobId } });
        if (
          !latest ||
          latest.status === "CANCELLED" ||
          latest.cancelRequested
        ) {
          logger.info("Job cancelled â€“ skipping processor", { jobId });
          return;
        }

        if (!workerRegistry.canProcess(claimed.type)) {
          await jobLifecycle.releaseToQueue(
            jobId,
            `worker ${config.WORKER_ID} lacks capability ${claimed.type}`,
          );
          return;
        }

        const processor = getProcessor(claimed.type);
        const result = await processor(claimed.payload);
        await jobLifecycle.complete(jobId, result, claimed.attempts, t0);
      } catch (err) {
        const message = err instanceof Error ? err.message : String(err);
        await jobLifecycle.fail(jobId, message, claimed.attempts);
      } finally {
        stopHeartbeat();
      }
    },
    { connection: redis, concurrency: 5 },
  );

  worker.on("completed", (job) => {
    logger.info("BullMQ job completed", { bullJobId: job.id });
  });
  worker.on("failed", (job, err) => {
    logger.error("BullMQ job failed", { bullJobId: job?.id, err });
  });

  async function shutdown(signal: string) {
    logger.info("Shutting down workerâ€¦", { signal });
    await workerRegistry.markOffline(signal);
    await worker.close();
    process.exit(0);
  }

  process.on("SIGTERM", () => void shutdown("SIGTERM"));
  process.on("SIGINT", () => void shutdown("SIGINT"));
}

main().catch((err) => {
  logger.error("Failed to start worker", err);
  process.exit(1);
});

import pino from "pino";
import { config } from "./index.js";

const baseLogger = pino({
  level: config.NODE_ENV === "development" ? "debug" : "info",
  ...(config.NODE_ENV === "development" && {
    transport: {
      target: "pino-pretty",
      options: {
        colorize: true,
        translateTime: "SYS:standard",
        ignore: "pid,hostname",
      },
    },
  }),
  base: {
    service: config.SERVICE_NAME,
  },
});

type LogMeta = Record<string, unknown> | Error | unknown;

type LogFn = {
  (msg: string, meta?: LogMeta): void;
  (meta: LogMeta, msg?: string): void;
};

const toMergingObject = (meta?: LogMeta): Record<string, unknown> => {
  if (meta === undefined || meta === null) {
    return {};
  }

  if (meta instanceof Error) {
    return {
      err: {
        message: meta.message,
        name: meta.name,
        stack: meta.stack,
      },
    };
  }

  if (typeof meta === "object") {
    return meta as Record<string, unknown>;
  }

  // primitive (string/number/boolean) â€“ wrap it
  return { value: meta };
};

const createLogFn = (level: "info" | "error" | "warn" | "debug"): LogFn => {
  return (msgOrMeta: string | LogMeta, maybeMetaOrMsg?: string | LogMeta) => {
    // Preferred: message first
    // logger.error("Error during graceful shutdown", error)
    // logger.info("Member added", { roomId })
    if (typeof msgOrMeta === "string") {
      const msg = msgOrMeta;
      const obj = toMergingObject(maybeMetaOrMsg);
      baseLogger[level](obj, msg);
      return;
    }

    // Also support: object/error first
    // logger.error(error, "Error during graceful shutdown")
    // logger.info({ roomId }, "Member added")
    const obj = toMergingObject(msgOrMeta);
    const msg = typeof maybeMetaOrMsg === "string" ? maybeMetaOrMsg : undefined;

    if (msg) {
      baseLogger[level](obj, msg);
    } else {
      baseLogger[level](obj);
    }
  };
};

export const logger = {
  info: createLogFn("info"),
  error: createLogFn("error"),
  warn: createLogFn("warn"),
  debug: createLogFn("debug"),
  child: baseLogger.child.bind(baseLogger),
};

      priority: bullPriority,
    },
  );

    priority: priority ?? "MEDIUM",
    bullPriority,
  });
}

import {
  calculateBackoffMs,
  generateId,
  NotFoundError,
  sleep,
} from "@job-platform/shared-utils";
import { JOB_EVENT_TYPES } from "@job-platform/shared-events";
import { TOPICS } from "@job-platform/shared-kafka";
import { Prisma, prisma } from "../infrastructure/prisma.js";
import { config } from "../config/index.js";
import { logger } from "../config/logger.js";
import { enqueueJob } from "../infrastructure/bullmq.js";

export class JobLifecycleService {
  async claim(jobId: string) {
    const leaseExpiresAt = new Date(Date.now() + config.LEASE_MS);
    const eventId = generateId();

    const claimed = await prisma.$transaction(async (tx) => {
      // Optimistic: only one worker can win QUEUED|RETRYING â†’ RUNNING
      const result = await tx.job.updateMany({
        where: {
          id: jobId,
          status: { in: ["QUEUED", "RETRYING"] },
        },
        data: {
          status: "RUNNING",
          attempts: { increment: 1 },
          leaseOwner: config.WORKER_ID,
          leaseExpiresAt,
          // First claim wins startedAt; retries keep original if already set.
          // Prisma updateMany can't do COALESCE easily â€” set only when we know
          // it's the first attempt by reading after update, or always set:
          startedAt: new Date(),
        },
      });

      if (result.count === 0) {
        return null;
      }

      const job = await tx.job.findUniqueOrThrow({ where: { id: jobId } });

      // Optional: if you want startedAt only on first claim, fix it here:
      // if (job.attempts > 1 && job.startedAt was overwritten) { ... }
      // For learning, overwriting startedAt on retry is acceptable.

      await tx.outbox.create({
        data: {
          id: generateId(),
          eventId,
          eventType: JOB_EVENT_TYPES.STARTED,
          topic: TOPICS.JOB_EVENTS,
          payload: {
            eventId,
            eventType: JOB_EVENT_TYPES.STARTED,
            occurredAt: new Date().toISOString(),
            jobId,
            jobType: job.type,
            tenantId: job.tenantId,
            payload: {
              workerId: config.WORKER_ID,
              attempt: job.attempts,
              leaseExpiresAt: leaseExpiresAt.toISOString(),
            },
          },
        },
      });

      return job;
    });

    if (!claimed) {
      // Lost the race or job not claimable
      const current = await prisma.job.findUnique({ where: { id: jobId } });
      if (!current) throw new NotFoundError("Job", jobId);
      logger.warn("Skip job â€“ not claimable (lost race or bad status)", {
        jobId,
        status: current.status,
      });
      throw new Error(`Job not claimable: ${current.status}`);
    }

    return {
      id: claimed.id,
      type: claimed.type,
      payload: claimed.payload,
      attempts: claimed.attempts,
    };
  }
  async complete(
    jobId: string,
    result: Record<string, unknown>,
    attempt: number,
    startedAtMs: number,
  ): Promise<void> {
    const eventId = generateId();
    const durationMs = Date.now() - startedAtMs;
    const resultJson = result as Prisma.InputJsonValue;

    const didComplete = await prisma.$transaction(async (tx) => {
      const updated = await tx.job.updateMany({
        where: {
          id: jobId,
          status: "RUNNING",
          leaseOwner: config.WORKER_ID,
        },
        data: {
          status: "COMPLETED",
          result: resultJson,
          error: null,
          leaseOwner: null,
          leaseExpiresAt: null,
          completedAt: new Date(),
        },
      });

      if (updated.count === 0) return false;

      const job = await tx.job.findUniqueOrThrow({ where: { id: jobId } });

      await tx.outbox.create({
        data: {
          id: generateId(),
          eventId,
          eventType: JOB_EVENT_TYPES.COMPLETED,
          topic: TOPICS.JOB_EVENTS,
          payload: {
            eventId,
            eventType: JOB_EVENT_TYPES.COMPLETED,
            occurredAt: new Date().toISOString(),
            jobId,
            jobType: job.type,
            tenantId: job.tenantId,
            payload: {
              workerId: config.WORKER_ID,
              attempt,
              resultJson,
              durationMs,
            },
          },
        },
      });

      return true;
    });

    if (!didComplete) {
      logger.warn("Complete skipped â€“ job not RUNNING under this worker", {
        jobId,
      });
      return;
    }

    logger.info("Job completed", { jobId, durationMs });
  }
  /**
   * Fail the job. If attempts remain â†’ RETRYING â†’ (backoff) â†’ QUEUED + enqueue.
   * If exhausted â†’ DEAD.
   *
   * Why domain-level retry (not BullMQ attempts)?
   * Postgres is source of truth for attempts/maxAttempts/status.
   * BullMQ is only the execution transport (attempts: 1).
   */

  /**
   * Fail the job. If attempts remain â†’ RETRYING â†’ (backoff) â†’ QUEUED + enqueue.
   * If exhausted â†’ DEAD.
   *
   * Concurrency: first transition uses
   *   where: { id, status: "RUNNING", leaseOwner: this worker }
   * so a reaper/other worker cannot be overwritten.
   */
  async fail(jobId: string, error: string, attempt: number): Promise<void> {
    const failedEventId = generateId();

    // 1) RUNNING â†’ FAILED (optimistic)
    const failed = await prisma.$transaction(async (tx) => {
      const updated = await tx.job.updateMany({
        where: {
          id: jobId,
          status: "RUNNING",
          leaseOwner: config.WORKER_ID,
        },
        data: {
          status: "FAILED",
          error,
          leaseOwner: null,
          leaseExpiresAt: null,
        },
      });

      if (updated.count === 0) {
        return null; // lost race â€” reaped, completed, or another worker
      }

      const job = await tx.job.findUniqueOrThrow({ where: { id: jobId } });
      const willRetry = attempt < job.maxAttempts;

      await tx.outbox.create({
        data: {
          id: generateId(),
          eventId: failedEventId,
          eventType: JOB_EVENT_TYPES.FAILED,
          topic: TOPICS.JOB_EVENTS,
          payload: {
            eventId: failedEventId,
            eventType: JOB_EVENT_TYPES.FAILED,
            occurredAt: new Date().toISOString(),
            jobId,
            jobType: job.type,
            tenantId: job.tenantId,
            payload: {
              workerId: config.WORKER_ID,
              attempt,
              error,
              willRetry,
            },
          },
        },
      });

      return { job, willRetry };
    });

    if (!failed) {
      logger.warn("Fail skipped â€“ job not RUNNING under this worker", {
        jobId,
      });
      return;
    }

    const { job, willRetry } = failed;

    logger.warn("Job failed", {
      jobId,
      error,
      willRetry,
      attempt,
      maxAttempts: job.maxAttempts,
    });

    if (!willRetry) {
      await this.markDead(jobId, job.type, job.tenantId, attempt, error);
      return;
    }

    // 2) FAILED â†’ RETRYING
    const backoffMs = calculateBackoffMs(attempt);
    const nextAttemptAt = new Date(Date.now() + backoffMs);
    const retryingEventId = generateId();

    const didRetry = await prisma.$transaction(async (tx) => {
      const updated = await tx.job.updateMany({
        where: {
          id: jobId,
          status: "FAILED",
        },
        data: {
          status: "RETRYING",
        },
      });

      if (updated.count === 0) {
        return false;
      }

      await tx.outbox.create({
        data: {
          id: generateId(),
          eventId: retryingEventId,
          eventType: JOB_EVENT_TYPES.RETRYING,
          topic: TOPICS.JOB_EVENTS,
          payload: {
            eventId: retryingEventId,
            eventType: JOB_EVENT_TYPES.RETRYING,
            occurredAt: new Date().toISOString(),
            jobId,
            jobType: job.type,
            tenantId: job.tenantId,
            payload: {
              attempt,
              nextAttemptAt: nextAttemptAt.toISOString(),
              error,
            },
          },
        },
      });

      return true;
    });

    if (!didRetry) {
      logger.warn("Retry transition skipped â€“ status changed", { jobId });
      return;
    }

    logger.info("Job entering retry", {
      jobId,
      attempt,
      backoffMs,
      nextAttemptAt: nextAttemptAt.toISOString(),
    });

    // 3) Backoff then RETRYING â†’ QUEUED + enqueue
    await sleep(backoffMs);
    await this.promoteRetryToQueued(jobId, job.type, job.tenantId);
  }

  private async markDead(
    jobId: string,
    jobType: string,
    tenantId: string | null,
    attempts: number,
    lastError: string,
  ): Promise<void> {
    const eventId = generateId();

    const didMark = await prisma.$transaction(async (tx) => {
      const updated = await tx.job.updateMany({
        where: {
          id: jobId,
          status: "FAILED",
        },
        data: {
          status: "DEAD",
        },
      });

      if (updated.count === 0) return false;

      await tx.outbox.create({
        data: {
          id: generateId(),
          eventId,
          eventType: JOB_EVENT_TYPES.DEAD,
          topic: TOPICS.JOB_EVENTS,
          payload: {
            eventId,
            eventType: JOB_EVENT_TYPES.DEAD,
            occurredAt: new Date().toISOString(),
            jobId,
            jobType,
            tenantId,
            payload: {
              attempts,
              lastError,
            },
          },
        },
      });

      return true;
    });

    if (!didMark) {
      logger.warn("markDead skipped â€“ status changed", { jobId });
      return;
    }

    logger.error("Job dead â€“ retries exhausted", {
      jobId,
      attempts,
      lastError,
    });
  }

  private async promoteRetryToQueued(
    jobId: string,
    jobType: string,
    tenantId: string | null,
  ): Promise<void> {
    const eventId = generateId();

    const promoted = await prisma.$transaction(async (tx) => {
      const updated = await tx.job.updateMany({
        where: {
          id: jobId,
          status: "RETRYING",
        },
        data: {
          status: "QUEUED",
        },
      });

      if (updated.count === 0) {
        return null;
      }

      const job = await tx.job.findUniqueOrThrow({ where: { id: jobId } });

      await tx.outbox.create({
        data: {
          id: generateId(),
          eventId,
          eventType: JOB_EVENT_TYPES.QUEUED,
          topic: TOPICS.JOB_EVENTS,
          payload: {
            eventId,
            eventType: JOB_EVENT_TYPES.QUEUED,
            occurredAt: new Date().toISOString(),
            jobId,
            jobType,
            tenantId,
            payload: {
              status: "QUEUED",
              priority: job.priority,
            },
          },
        },
      });

      return job;
    });

    if (!promoted) {
      logger.debug("promoteRetryToQueued skipped â€“ status changed", { jobId });
      return;
    }

    try {
      await enqueueJob(jobId,promoted.priority);
    } catch (err) {
      logger.error("Retry promote: DB is QUEUED but enqueue failed", {
        jobId,
        err,
      });
    }

    logger.info("Job requeued after backoff", { jobId });
  }

  /**
   * Worker cannot run this type: put back to QUEUED and re-enqueue.
   * Does not count as a failure / attempt.
   */
  async releaseToQueue(jobId: string, reason: string): Promise<void> {
    const eventId = generateId();
  
    const released = await prisma.$transaction(async (tx) => {
      const updated = await tx.job.updateMany({
        where: {
          id: jobId,
          status: "RUNNING",
          leaseOwner: config.WORKER_ID,
        },
        data: {
          status: "QUEUED",
          leaseOwner: null,
          leaseExpiresAt: null,
          // roll back the attempt increment from claim
          attempts: { decrement: 1 },
        },
      });
  
      if (updated.count === 0) return null;
  
      const job = await tx.job.findUniqueOrThrow({ where: { id: jobId } });
  
      // Clamp attempts at 0
      if (job.attempts < 0) {
        await tx.job.update({
          where: { id: jobId },
          data: { attempts: 0 },
        });
      }
  
      await tx.outbox.create({
        data: {
          id: generateId(),
          eventId,
          eventType: JOB_EVENT_TYPES.QUEUED,
          topic: TOPICS.JOB_EVENTS,
          payload: {
            eventId,
            eventType: JOB_EVENT_TYPES.QUEUED,
            occurredAt: new Date().toISOString(),
            jobId,
            jobType: job.type,
            tenantId: job.tenantId,
            payload: {
              status: "QUEUED",
              priority: job.priority,
              releaseReason: reason,
            },
          },
        },
      });
  
      return job;
    });
  
    if (!released) {
      logger.warn("releaseToQueue skipped â€“ not owned RUNNING", { jobId });
      return;
    }
  
    try {
      await enqueueJob(jobId,released.priority);
    } catch (err) {
      logger.error("releaseToQueue: DB QUEUED but enqueue failed", { jobId, err });
    }
  
    logger.info("Job released to queue (unsupported type)", { jobId, reason });
  }
}

/**
 * Renews leaseExpiresAt while a job is still being processed.
 * Why? If the worker dies, the reaper can reclaim after leaseExpiresAt.
 * We renew at ~1/3 of LEASE_MS so a few missed ticks still leave a valid lease.
 */

import { config } from "../config/index.js";
import { logger } from "../config/logger.js";
import { prisma } from "../infrastructure/prisma.js";

export function startLeaseHeartbeat(jobId: string): () => void {
  const intervalMs = Math.max(1_000, Math.floor(config.LEASE_MS / 3));
  const timer = setInterval(async () => {
    try {
      const nextExpiry = new Date(Date.now() + config.LEASE_MS);
      const result = await prisma.job.updateMany({
        where: { id: jobId, status: "RUNNING", leaseOwner: config.WORKER_ID },
        data: {
          leaseExpiresAt: nextExpiry,
        },
      });
      if (result.count === 0) {
        // Job was completed/failed/reaped â€” stop renewing.
        logger.debug("Lease heartbeat: job no longer owned, stopping", {
          jobId,
        });
        clearInterval(timer);
        return;
      }

      logger.debug("Lease renewed", {
        jobId,
        leaseExpiresAt: nextExpiry.toISOString(),
      });
    } catch (err) {
      logger.warn("Lease heartbeat failed", { jobId, err });
      // Don't clear the timer â€” transient DB blips shouldn't stop renewal.
    }
  }, intervalMs);
  // Return stop function so the caller can clean up when the job finishes.
  return () => clearInterval(timer);
}

const ALLOWED: Record<JobStatus, readonly JobStatus[]> = {
  PENDING: [JobStatus.QUEUED, JobStatus.SCHEDULED, JobStatus.CANCELLED],
  SCHEDULED: [JobStatus.QUEUED, JobStatus.CANCELLED],
  QUEUED: [JobStatus.RUNNING, JobStatus.CANCELLED],
  RUNNING: [
    JobStatus.COMPLETED,
    JobStatus.FAILED,
    JobStatus.CANCELLED,
    JobStatus.QUEUED, // reaper lease recovery only
  ],
  FAILED: [JobStatus.RETRYING, JobStatus.DEAD,JobStatus.QUEUED],// admin retry
  RETRYING: [JobStatus.QUEUED,JobStatus.CANCELLED],
  COMPLETED: [],
  DEAD: [JobStatus.QUEUED,JobStatus.CANCELLED],// admin retry
  CANCELLED: [],
};

export function assertTransition(from: JobStatus, to: JobStatus): void {
  if (!ALLOWED[from].includes(to)) {
    throw new InvalidStateTransitionError(from, to);
  }
}

import { generateId } from "@job-platform/shared-utils";
import { prisma } from "../infrastructure/prisma.js";
import { config } from "../config/index.js";
import { WORKER_EVENT_TYPES } from "@job-platform/shared-events";
import { TOPICS } from "@job-platform/shared-kafka";
import { logger } from "../config/logger.js";

/**
 * Registers this process in Postgres and keeps lastHeartbeatAt fresh.
 * Why DB? Same source of truth as jobs; ops can query without Redis-only state.
 */
export class WorkerRegistryService {
  private timer: ReturnType<typeof setInterval> | null = null;

  async register(): Promise<void> {
    const now = new Date();
    const eventId = generateId();
    await prisma.$transaction(async (tx) => {
      await tx.worker.upsert({
        where: { id: config.WORKER_ID },
        create: {
          id: config.WORKER_ID,
          capabilities: [...config.WORKER_CAPABILITIES],
          registeredAt: now,
          lastHeartbeatAt: now,
          status: "ONLINE",
        },
        update: {
          capabilities: [...config.WORKER_CAPABILITIES],
          status: "ONLINE",
          lastHeartbeatAt: now,
        },
      });
      await tx.outbox.create({
        data: {
          id: generateId(),
          eventId,
          eventType: WORKER_EVENT_TYPES.REGISTERED,
          topic: TOPICS.WORKER_EVENTS,
          payload: {
            eventId,
            eventType: WORKER_EVENT_TYPES.REGISTERED,
            occurredAt: now.toISOString(),
            workerId: config.WORKER_ID,
            payload: {
              capabilities: config.WORKER_CAPABILITIES,
            },
          },
        },
      });
    });
    logger.info("Worker registered", {
      workerId: config.WORKER_ID,
      capabilities: config.WORKER_CAPABILITIES,
    });
  }

  startHeartbeat(intervalMs = config.WORKER_HEARTBEAT_MS): void {
      if (this.timer) return;
  
      this.timer = setInterval(() => {
        void this.beat();
      }, intervalMs);
  
      logger.info("Worker heartbeat started", { intervalMs });
    }
  
    stopHeartbeat(): void {
      if (this.timer) {
        clearInterval(this.timer);
        this.timer = null;
      }
    }
    private async beat(): Promise<void> {
        try {
          const now = new Date();
          await prisma.worker.updateMany({
            where: { id: config.WORKER_ID },
            data: {
              status: "ONLINE",
              lastHeartbeatAt: now,
            },
          });
          // Optional: outbox HEARTBEAT every N beats to avoid noise â€” skip for Phase 6
        } catch (err) {
          logger.warn("Worker heartbeat failed", { err });
        }
    }
    async markOffline(reason?: string): Promise<void> {
        this.stopHeartbeat();
        const eventId = generateId();
        const now = new Date();
    
        try {
          await prisma.$transaction(async (tx) => {
            await tx.worker.updateMany({
              where: { id: config.WORKER_ID },
              data: { status: "OFFLINE", lastHeartbeatAt: now },
            });
    
            await tx.outbox.create({
              data: {
                id: generateId(),
                eventId,
                eventType: WORKER_EVENT_TYPES.OFFLINE,
                topic: TOPICS.WORKER_EVENTS,
                payload: {
                  eventId,
                  eventType: WORKER_EVENT_TYPES.OFFLINE,
                  occurredAt: now.toISOString(),
                  workerId: config.WORKER_ID,
                  payload: { reason: reason ?? "shutdown" },
                },
              },
            });
          });
          logger.info("Worker marked offline", { workerId: config.WORKER_ID });
        } catch (err) {
          logger.warn("Failed to mark worker offline", { err });
        }
    }
    canProcess(jobType: string): boolean {
        return config.WORKER_CAPABILITIES.includes(jobType);
      }
}
export const workerRegistry = new WorkerRegistryService();

import type { JobPriority, JobStatus, JobType, Uuid } from "@job-platform/shared-types";

export const JOB_EVENT_TYPES = {
  CREATED: "job.created",
  QUEUED: "job.queued",
  STARTED: "job.started",
  COMPLETED: "job.completed",
  FAILED: "job.failed",
  RETRYING: "job.retrying",
  DEAD: "job.dead",
  CANCELLED: "job.cancelled",
  LEASE_EXPIRED: "job.lease_expired",
  SCHEDULED: "job.scheduled",
} as const;

export type JobEventType = (typeof JOB_EVENT_TYPES)[keyof typeof JOB_EVENT_TYPES];

export interface JobEventBase {
  eventId: Uuid;
  eventType: JobEventType;
  occurredAt: string;
  jobId: Uuid;
  jobType: JobType | string;
  tenantId?: string | null;
}

export interface JobCreatedEvent extends JobEventBase {
  eventType: typeof JOB_EVENT_TYPES.CREATED;
  payload: {
    status: JobStatus;
    priority: JobPriority;
    scheduledAt?: string | null;
    idempotencyKey?: string | null;
    maxAttempts: number;
  };
}

export interface JobQueuedEvent extends JobEventBase {
  eventType: typeof JOB_EVENT_TYPES.QUEUED;
  payload: {
    status: JobStatus;
    priority: JobPriority;
  };
}

export interface JobStartedEvent extends JobEventBase {
  eventType: typeof JOB_EVENT_TYPES.STARTED;
  payload: {
    workerId: string;
    attempt: number;
    leaseExpiresAt: string;
  };
}

export interface JobCompletedEvent extends JobEventBase {
  eventType: typeof JOB_EVENT_TYPES.COMPLETED;
  payload: {
    workerId: string;
    attempt: number;
    result?: Record<string, unknown> | null;
    durationMs: number;
  };
}

export interface JobFailedEvent extends JobEventBase {
  eventType: typeof JOB_EVENT_TYPES.FAILED;
  payload: {
    workerId?: string | null;
    attempt: number;
    error: string;
    willRetry: boolean;
  };
}

export interface JobRetryingEvent extends JobEventBase {
  eventType: typeof JOB_EVENT_TYPES.RETRYING;
  payload: {
    attempt: number;
    nextAttemptAt: string;
    error: string;
  };
}

export interface JobDeadEvent extends JobEventBase {
  eventType: typeof JOB_EVENT_TYPES.DEAD;
  payload: {
    attempts: number;
    lastError: string;
  };
}

export interface JobCancelledEvent extends JobEventBase {
  eventType: typeof JOB_EVENT_TYPES.CANCELLED;
  payload: {
    reason?: string | null;
  };
}

export interface JobLeaseExpiredEvent extends JobEventBase {
  eventType: typeof JOB_EVENT_TYPES.LEASE_EXPIRED;
  payload: {
    previousWorkerId: string;
    leaseExpiredAt: string;
  };
}

export interface JobScheduledEvent extends JobEventBase {
  eventType: typeof JOB_EVENT_TYPES.SCHEDULED;
  payload: {
    scheduledAt: string;
  };
}

export type JobDomainEvent =
  | JobCreatedEvent
  | JobQueuedEvent
  | JobStartedEvent
  | JobCompletedEvent
  | JobFailedEvent
  | JobRetryingEvent
  | JobDeadEvent
  | JobCancelledEvent
  | JobLeaseExpiredEvent
  | JobScheduledEvent;

import { Kafka, KafkaConfig, logLevel } from "kafkajs";

export interface CreateKafkaClientOptions {
  clientId: string;
  brokers: string[];
  logLevel?: logLevel;
}

export function createKafkaClient(options: CreateKafkaClientOptions): Kafka {
  const config: KafkaConfig = {
    brokers: options.brokers,
    clientId: options.clientId,
    logLevel: options.logLevel ?? logLevel.WARN,
    retry: {
      initialRetryTime: 300,
      retries: 8,
    },
  };
  return new Kafka(config);
}

import type {
  Kafka,
  Consumer,
  EachMessagePayload,
  ConsumerSubscribeTopics,
} from "kafkajs";

export type MessageHandler = (payload: EachMessagePayload) => Promise<void>;

export class KafkaConsumer {
  private consumer: Consumer;
  private connected = false;

  constructor(
    kafka: Kafka,
    private readonly groupId: string,
  ) {
    this.consumer = kafka.consumer({
      groupId: this.groupId,
      sessionTimeout: 30_000,
      heartbeatInterval: 3_000,
    });
  }

  async connect(): Promise<void> {
    if (this.connected) return;
    await this.consumer.connect();
    this.connected = true;
  }

  async disconnect(): Promise<void> {
    if (!this.connected) return;
    await this.consumer.disconnect();
    this.connected = false;
  }

  async subscribe(topics: ConsumerSubscribeTopics): Promise<void> {
    if (!this.connected) await this.connect();
    await this.consumer.subscribe(topics);
  }

  async run(handler: MessageHandler): Promise<void> {
    if (!this.connected) await this.connect();
    await this.consumer.run({ eachMessage: handler });
  }
}

import type { Kafka, Producer, ProducerRecord, RecordMetadata } from "kafkajs";

export class KafkaProducer {
  private producer: Producer;
  private connected = false;

  constructor(kafka: Kafka) {
    this.producer = kafka.producer({
      allowAutoTopicCreation: true,
      idempotent: true,
    });
  }

  async connect(): Promise<void> {
    if (this.connected) return;
    await this.producer.connect();
    this.connected = true;
  }

  async disconnect(): Promise<void> {
    if (!this.connected) return;
    await this.producer.disconnect();
    this.connected = false;
  }

  async send(record: ProducerRecord): Promise<RecordMetadata[]> {
    if (!this.connected) await this.connect();
    return this.producer.send(record);
  }

  async sendBatch(records: ProducerRecord[]): Promise<RecordMetadata[]> {
    if (!this.connected) await this.connect();
    const result = await this.producer.sendBatch({ topicMessages: records });
    return result.flat();
  }
}

import type { Request, Response, NextFunction, RequestHandler } from "express";

export function asyncHandler(
  fn: (req: Request, res: Response, next: NextFunction) => Promise<unknown>,
): RequestHandler {
  return (req, res, next) => {
    Promise.resolve(fn(req, res, next)).catch(next);
  };
}

export function generateId(): Uuid {
  return uuidv4() as Uuid;
}

export function sleep(ms: number): Promise<void> {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

export function calculateBackoffMs(
  attempt: number,
  baseMs = 1000,
  capMs = 60_000,
): number {
  const exp = Math.min(capMs, baseMs * Math.pow(2, attempt));
  return Math.floor(Math.random() * exp);
}

export type DomainPriority = "LOW" | "HIGH" | "MEDIUM" | string

const PRIORITY_MAP: Record<string, number> = {
  HIGH: 1,
  MEDIUM: 5,
  LOW:10
}

export const DEFAULT_BULLMQ_PRIORITY = 5;

export function mapPriorityToBullmq(priority?: DomainPriority | null): number {
  if (!priority) return DEFAULT_BULLMQ_PRIORITY;
  return PRIORITY_MAP[priority] ?? DEFAULT_BULLMQ_PRIORITY;
}
